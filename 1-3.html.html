١-٣ مساحة القطاع

طبّق أ = ½ نق٢ هـ بالراديان، ثم مساحة القطعة = قطاع − مثلث. π تُكتب يمين الرقم مثل ٣π، والجذر يمين الرقم ومفتوح عليه مثل ٣√.

نتيجتك: ٠ / ٠

π القوانين

مساحة القطاعأ = ½ نق٢ هـ  (هـ بالراديان)
محيط القطاعمحيط = ٢نق + ن = نق(٢ + هـ)
مساحة القطعةأ = ½ نق٢ (هـ − جا هـ)
جيب التمامجتا هـ = (نق٢ + نق٢ − وتر٢) ÷ (٢نق٢)
أ = مساحة القطاع ن = طول القوس نق = نصف القطر هـ = الزاوية بالراديان جا / جتا = الجيب / جيب التمام

اكتب الكسر في خانتين. π تكون يمين الرقم مثل ٣π. الجذر يُكتب يمين الرقم ومفتوحًا عليه مثل ٣√. التقريب إلى ٣ أرقام معنوية عند الحاجة.

تمارين ١-٣ • مساحة القطاع

١ من المساحة إلى نصف القطر

هـ = ١٫٢ راد ، المساحة = ٥٤ سم٢. أوجد نق.
القانون: أ = ½ نق٢ هـ

نق ≈
٥٤ = ½ × نق٢ × ١٫٢ = ٠٫٦ نق٢
  1. ٥٤ = ٠٫٦ نق٢
  2. نق٢ = ٥٤ ÷ ٠٫٦ = ٩٠
  3. نق = ٩٠√ = ٣١٠√ ≈ ٩٫٤٩ سم

٢ من المحيط إلى المساحة

محيط القطاع = ٢٨ سم ، والزاوية ١٫٦ راد. أوجد المساحة.
أولًا أوجد نق من المحيط ثم أ = ½ نق٢ هـ

المساحة ≈
٢٨ = نق(٢+١٫٦) = ٣٫٦ نق
  1. نق = ٢٨ ÷ ٣٫٦ ≈ ٧٫٧٧٨ سم
  2. أ = ½ × (٧٫٧٧٨)٢ × ١٫٦ ≈ ٠٫٨ × ٦٠٫٤٩ ≈ ٤٨٫٤ سم٢

٣ زاوية بدلالة π

هـ = π٠٫٩ ، المساحة = ١٨٠ سم٢. أوجد نق.

نق ≈
١٨٠ = ½ نق٢ × ٠٫٩π = ٠٫٤٥π نق٢
  1. ١٨٠ = (٩π/٢٠) نق٢
  2. نق٢ = ٣٦٠٠ ÷ (٩π) = ٤٠٠/π
  3. نق = ٢٠/π√ ≈ ١١٫٣ سم

٤ المنطقة المظللة

قطاع زاويته ٤٥° ونصف قطره ٦ سم (الشكل المظلل قطاع).
أ = (هـ°/٣٦٠)π نق٢ أو حوّل ٤٥° إلى π/٤ ثم أ = ½ نق٢ هـ

المساحة ≈
٤٥° = π/٤ ، أ = ½ × ٣٦ × π/٤ = ٩π/٢
  1. ٤٥° = π٤
  2. أ = ½ × ٦٢ × π/٤ = ١٨ × π/٤ = ٩π٢
  3. أو ٩π/٢ ≈ ١٤٫١ سم٢

٥ فرق مساحتي القطاعين

نق = ٥ سم. الفرق بين مساحة القطاع الأكبر والأصغر = ١٥ سم٢. أوجد الزاوية المنفرجة ل م ك.
إذا كانت هـ الزاوية الصغرى: الفرق = نق٢ (π − هـ)

الزاوية المنفرجة ≈
٢٥(π − هـ) = ١٥ ثم هـ = π − ٠٫٦
  1. الفرق = ½ نق٢(٢π−هـ) − ½ نق٢هـ = نق٢(π − هـ)
  2. ٢٥(π − هـ) = ١٥
  3. π − هـ = ٠٫٦
  4. هـ = π − ٠٫٦ ≈ ٢٫٥٤١٦ ≈ ٢٫٥٤ راد (منفرجة لأن ٢٫٥٤ راد ≈ ١٤٦°)

٦ مساحة القطعة المظللة

أوجد مساحة المنطقة المظللة حيث م مركز الدائرة، نق = ١٣ سم، والوتر أ ب = ١٠ سم.
القوانين: جتا هـ = (نق٢+نق٢−وتر٢) ÷ (٢ نق نق)  •  مساحة القطعة = ½ نق٢ (هـ − جا هـ)

مساحة القطعة المظللة ≈
من قانون جيب التمام أوجد جتا هـ ثم جا هـ، ثم القطعة = قطاع − مثلث.
القانون: جتا هـ = (أ٢+ب٢−ج٢)/(٢أب) ، أقطعة = ½ نق٢(هـ − جا هـ)
  1. جتا هـ = (١٣٢ + ١٣٢ − ١٠٢) ÷ (٢×١٣×١٣) = (١٦٩+١٦٩−١٠٠)/٣٣٨ = ٢٣٨/٣٣٨ = ١١٩/١٦٩
  2. جا هـ = ١ − (١١٩/١٦٩)٢√ = ١٤٤٠٠√/١٦٩ = ١٢٠/١٦٩
  3. مساحة المثلث م أ ب = ½ × ١٣ × ١٣ × (١٢٠/١٦٩) = (١٦٩/٢)×(١٢٠/١٦٩) = ٦٠ سم٢
  4. هـ = قوس جتا(١١٩/١٦٩) ≈ ٠٫٧٩٥٦ راد
  5. مساحة القطاع = ½ × ١٦٩ × ٠٫٧٩٥٦ ≈ ٦٧٫٢٣ سم٢
  6. القطعة = ٦٧٫٢٣ − ٦٠ ≈ ٧٫٢٣ سم٢

٧ قطعة مساحتها ربع الدائرة

وتر يقابل الزاوية المركزية هـ، ويقطع قطعة دائرية مساحتها ¼ المساحة الإجمالية للدائرة.
القانون: مساحة القطعة = ½ نق٢ (هـ − جا هـ)

أ) بيّن أن هـ − جا هـ = π/٢ (اكتب π/٢ أو نعم بعد التبيين)
ب) هـ ≈ (٣ أرقام معنوية)
ساوِ ½ نق٢(هـ−جا هـ) مع ¼ π نق٢ ثم اختصر نق٢.
  1. ½ نق٢ (هـ − جا هـ) = ¼ π نق٢
  2. اقسم الطرفين على ½ نق٢: هـ − جا هـ = π/٢
  3. إذن هـ − جا هـ = π/٢
  4. حلّ عدديًا: جرّب هـ = ٢٫٣١ ، جا(٢٫٣١) ≈ ٠٫٧٤١٦
  5. ٢٫٣١ − ٠٫٧٤١٦ = ١٫٥٦٨٤ ≈ π/٢ = ١٫٥٧٠٨
  6. إلى ٣ أرقام معنوية: هـ = ٢٫٣١ راد

٨ قطاع ونصف دائرة على القطر

ا ب ع قطاع في دائرة نصف قطرها ١٠ سم ومركزها ا، وهـ = ق(ب ا ع). مساحة نصف الدائرة التي قطرها ا ع تساوي ضعف مساحة القطاع ا ب ع.
القانون: أقطاع = ½ نق٢ هـ ، أنصف دائرة = ½ π (القطر/٢)٢

أ) هـ بدلالة π
ب) محيط كامل الشكل بدلالة π
قطر ا ع = ١٠ لذا نصف قطر نصف الدائرة = ٥. ساوِ ١٢٫٥π = ٢ × (٥٠ هـ).
  1. ا ع = نق = ١٠ سم، فنصف قطر نصف الدائرة على القطر ا ع = ٥ سم
  2. مساحة نصف الدائرة = ½ π × ٥٢ = ٢٥π/٢ أو ١٢٫٥π
  3. مساحة القطاع = ½ × ١٠٢ × هـ = ٥٠ هـ
  4. ١٢٫٥π = ٢ × ٥٠ هـ = ١٠٠ هـ
  5. هـ = ١٢٫٥π ÷ ١٠٠ = π/٨
  6. المحيط = الضلع ا ب + قوس ب ع + قوس نصف الدائرة ع ا
  7. ا ب = ١٠ ، قوس ب ع = ١٠ × π/٨ = ٥π/٤ ، قوس نصف الدائرة = π×٥ = ٥π
  8. المحيط = ١٠ + ٥π/٤ + ٥π = ١٠ + ٥π/٤ + ٢٠π/٤ = ١٠ + ٢٥π/٤

٩ قطعة معدنية من مثلث متساوي الأضلاع

مثلث متساوي الأضلاع ا ب ع طول ضلعه ٢ سم. تمر دائرة مركزها ا بالنقطتين ب ، ع، ودائرة مركزها ب بالنقطتين ا ، ع، ودائرة مركزها ع بالنقطتين ا ، ب. القطعة المعدنية هي المنطقة الداخلية المشتركة (مثلث رولو).
القانون: كل قوس يقابل ٦٠° = π/٣ ، المحيط = ٣ × (نق × هـ) ، المساحة = مساحة المثلث + ٣ قطع

المحيط بدلالة π
مساحة أحد الوجهين
كل قوس ٦٠° بنصف قطر ٢. القطعة الواحدة = ½×٤×(π/٣ − ٣√/٢).
  1. كل زاوية في المثلث ٦٠° = π/٣ ، ونصف قطر كل دائرة = ٢ سم
  2. طول القوس الواحد = ٢ × π/٣ = ٢π/٣
  3. المحيط = ٣ × ٢π/٣ = ٢π سم
  4. مساحة المثلث = (٣√/٤)×٢٢ = ٣√
  5. قطعة واحدة = ½ × ٤ × (π/٣ − جا π/٣) = ٢(π/٣ − ٣√/٢) = ٢π/٣ − ٣√
  6. ثلاث قطع = ٢π − ٣٣√
  7. المساحة الكلية = ٣√ + ٢π − ٣٣√ = ٢π − ٢٣√ = ٢(π − ٣√) سم٢

١٠ أكبر مساحة لقطاع محيطه ثابت

دائرة مركزها و ونصف قطرها نق. قطاع دائري محيطه ٦ سم وزاويته هـ.
القانون: محيط القطاع = نق(٢ + هـ) = ٦ ، المساحة أ = ½ نق٢ هـ

أ) مساحة القطاع بدلالة نق (بعد تبيين هـ = ٦/نق − ٢)
ب) نق الذي يعطي أكبر مساحة
ب) هـ المناظرة
من ٦ = نق(٢+هـ) نحصل على هـ. عوّض في أ ثم اكمل المربع أو اشتق.
  1. محيط القطاع = ٢نق + نق هـ = ٦
  2. نق(٢ + هـ) = ٦ ⇒ هـ = ٦/نق − ٢
  3. أ = ½ نق٢ (٦/نق − ٢) = ½(٦ نق − ٢ نق٢) = ٣ نق − نق٢
  4. أ = −(نق٢ − ٣ نق) = −(نق − ٣/٢)٢ + ٩/٤
  5. أكبر مساحة عندما نق = ٣/٢ سم ، وأكبر مساحة = ٩/٤ سم٢
  6. هـ = ٦ ÷ (٣/٢) − ٢ = ٤ − ٢ = ٢ راد
