١-١ الراديان • تمارين تفاعلية

حوّل بين الدرجة والراديان، استخدم القيم المثلثية الدقيقة، ثم طبّق قانون الجيب التمام على مثلث وسلم مائل.

نتيجتك: ٠ / ٠

ستتعلّم في هذا الدرس

  1. ١-١ تحويل القياس بين الراديان والدرجة.
  2. استخدام الصيغة θ(راديان) = θ(درجة) × π ÷ ١٨٠ وعكسها.
  3. إيجاد القيم الدقيقة لـ جا، جتا، ظا عند الزوايا الخاصة.
  4. التقريب إلى ٣ أرقام معنوية، مع الانتباه لوضع الآلة الحاسبة (راديان / درجة).

π التذكير السريع

درجة → راديانθراد = θدرجة × π١٨٠
راديان → درجةθدرجة = θراد × ١٨٠π
دورة كاملة٣٦٠° = ٢π راد ، ١٨٠° = π راد
قانون الجيب التمامأ٢ = ب٢ + جـ٢ − ٢ب جـ جتا (الزاوية)

اكتب الكسر في خانتين: البسط فوق والمقام تحت. π تكون يسار الرقم مثل π٣ وليس ٣π. الجذر يُكتب يمين الرقم ومفتوحًا باتجاهه مثل ٣√ وليس √٣.

١ حوّل من الدرجات إلى الراديان بدلالة π

حوّل قياس كل زاوية إلى الراديان، واكتب الناتج بدلالة π.
القانون: θراد = θدرجة × π١٨٠ ثم اختصر الكسر. تذكّر: ١٨٠° = π ، ٣٦٠° = π٢.

اضرب في π١٨٠ ثم اختصر الكسر. مثال: ٩٠° = π٩٠١٨٠ = π٢.
القانون المستخدم: θراد = θ° × π١٨٠ ثم نختصر البسط والمقام على القاسم المشترك الأكبر.
  1. أ) ٩٠° = π٩٠١٨٠ = اختصر ÷٩٠ = π٢
  2. ب) ١٤٠° = π١٤٠١٨٠ = اختصر ÷٢٠ = π٧٩
  3. ج) ٤٥° = π٤٥١٨٠ = اختصر ÷٤٥ = π٤
  4. د) ٣٠° = π٣٠١٨٠ = اختصر ÷٣٠ = π٦
  5. هـ) ٧٢° = π٧٢١٨٠ = اختصر ÷٣٦ = π٢٥
  6. و) ١٨° = π١٨١٨٠ = اختصر ÷١٨ = π١٠
  7. ز) ١٢٠° = π١٢٠١٨٠ = اختصر ÷٦٠ = π٢٣
  8. ح) ٢٢½° = ٢٢٫٥° = π٢٢٫٥١٨٠ = π٤٥٣٦٠ = اختصر ÷٤٥ = π٨
  9. ط) ٧٢٠° = دورتان كاملتان = ٢ × π٢ = π٤ ، أو π٧٢٠١٨٠ = π٤
  10. ي) ٤٢٠° = π٤٢٠١٨٠ = اختصر ÷٦٠ = π٧٣
  11. ك) ٢٧٠° = π٢٧٠١٨٠ = اختصر ÷٩٠ = π٣٢
  12. ل) ١° = π١١٨٠ = π١٨٠

حوّل كل قياس إلى درجات.
القانون: θدرجة = θراد × ١٨٠π يُلغى π ثم تُختصر. مثال: π٥ × ١٨٠π = ٣٦°.

اضرب في ١٨٠π فيُلغى π. مثال: π٥ × ١٨٠π = ٣٦°.
القانون: θ° = θراد × ١٨٠π
  1. أ) π١٨ × ١٨٠π = ١٨٠÷١٨ = ١٠°
  2. ب) π٢٠ × ١٨٠/π = ١٨٠÷٢٠ = ٩°
  3. ج) π٥ × ١٨٠/π = ١٨٠÷٥ = ٣٦°
  4. د) π٨ × ١٨٠/π = ١٨٠÷٨ = ٢٢٫٥°
  5. هـ) π٩ × ١٨٠/π = ١٨٠÷٩ = ٢٠°
  6. و) π٢٣ × ١٨٠/π = (٢×١٨٠)÷٣ = ٣٦٠÷٣ = ١٢٠°
  7. ز) π٥٨ × ١٨٠/π = (٥×١٨٠)÷٨ = ٩٠٠÷٨ = ١١٢٫٥°
  8. ح) π٣٥ × ١٨٠/π = (٣×١٨٠)÷٥ = ٥٤٠÷٥ = ١٠٨°
  9. ط) π٤٥ × ١٨٠/π = ١٨٠÷٤٥ = ٤°
  10. ي) π٦ × ١٨٠/π = ٦×١٨٠ = ١٠٨٠° (ثلاث دورات كاملة)
  11. ك) −π٢ × ١٨٠/π = −١٨٠÷٢ = −٩٠°
  12. ل) π٥١٨ × ١٨٠/π = (٥×١٨٠)÷١٨ = ٩٠٠÷١٨ = ٥٠°

٣ القيم المثلثية الدقيقة

اكتب القيمة بالضبط (بدون آلة حاسبة). يمكنك استخدام √ أو كلمة جذر.
القانون: حوّل الراديان إلى درجة أولًا: θ° = θراد × ١٨٠/π ثم استخدم القيم الخاصة: جا ٣٠=١/٢ ، جا ٤٥=√٢/٢ ، جا ٦٠=√٣/٢ ، جا ٩٠=١. جتا ٣٠=√٣/٢ ، جتا ٤٥=√٢/٢ ، جتا ٦٠=١/٢ ، جتا ٩٠=٠. ظا ٣٠=١/√٣ ، ظا ٤٥=١ ، ظا ٦٠=√٣. انتبه لإشارة الربع (موجب / سالب).

حوّل الراديان إلى درجة أولًا إن احتجت: π٣ = ٦٠° ، π٢٣ = ١٢٠° ، π٧٦ = ٢١٠° ، π٣٢ = ٢٧٠° ، π٧٤ = ٣١٥° ، π٥٣ = ٣٠٠°.
الخطوات: حوّل إلى درجة، حدّد الربع والإشارة، ثم استخدم القيمة الدقيقة للزاوية المرجعية.
  1. أ) π٣ × ١٨٠/π = ٦٠° (الربع الأول، جا موجبة) ← جا ٦٠° = ٣√٢
  2. ب) π٤ = ٤٥° ← جتا ٤٥° = ٢√٢
  3. ج) π٦ = ٣٠° ← ظا ٣٠° = ١٣√ أو بعد الضرب في √٣/√٣: ٣√٣
  4. د) π٣٢ = ٢٧٠° ← جتا ٢٧٠° = جتا(٣٦٠°−٩٠°) = ٠
  5. هـ) π٧٤ = ٣١٥° = ٣٦٠°−٤٥° (الربع الرابع، جا سالبة) ← جا ٣١٥° = −جا ٤٥° = −٢√٢
  6. و) π٧٦ = ٢١٠° = ١٨٠°+٣٠° (الربع الثالث، جتا سالبة) ← جتا ٢١٠° = −جتا ٣٠° = −٣√٢
  7. ز) π٥٣ = ٣٠٠° = ٣٦٠°−٦٠° (الربع الرابع، ظا سالبة) ← ظا ٣٠٠° = −ظا ٦٠° = −٣√
  8. ح) π٢٣ = ١٢٠° = ١٨٠°−٦٠° (الربع الثاني، جا موجبة) ← جا ١٢٠° = جا ٦٠° = ٣√٢

٤ إلى الراديان • ٣ أرقام معنوية

استخدم π ≈ ٣٫١٤١٦ ثم قرّب الناتج إلى أقرب ٣ أرقام معنوية.
القانون: θراد = θ° × π ÷ ١٨٠. ثلاثة أرقام معنوية تعني مثل ٥٫٠٣ و ٠٫٣٤٩ (لا تحسب الأصفار على يسار الرقم معنوية).

θراد = θ° × π١٨٠. ثلاثة أرقام معنوية تعني مثل ٥٫٠٣ و ٠٫٣٤٩.
القانون: θراد = θ° × π ÷ ١٨٠ ، π ≈ ٣٫١٤١٦
  1. أ-١) ٢٨٨ × ٣٫١٤١٦ ÷ ١٨٠ = ٩٠٤٫٧٨٠٨ ÷ ١٨٠ = ٥٫٠٢٦٥٦ ≈ ٥٫٠٣ راديان (٣ أرقام معنوية)
  2. أ-٢) ٢٠ × ٣٫١٤١٦ ÷ ١٨٠ = ٦٢٫٨٣٢ ÷ ١٨٠ = ٠٫٣٤٩٠٦ ≈ ٠٫٣٤٩ راديان
  3. ب-١) ٢٧٠° = π٣٢ = ١٫٥ × ٣٫١٤١٦ = ٤٫٧١٢٤ ≈ ٤٫٧١ راديان
  4. ب-٢) ٩٠° = π٢ = ٣٫١٤١٦ ÷ ٢ = ١٫٥٧٠٨ ≈ ١٫٥٧ راديان

٦ إلى الدرجات • ٣ أرقام معنوية

القانون: θ° = θراد × ١٨٠ ÷ π ، π ≈ ٣٫١٤١٦ ثم قرّب إلى ٣ أرقام معنوية.

اضرب في ١٨٠π. ١٫٥ × ١٨٠π ≈ ٨٥٫٩°.
القانون: θ° = θراد × ١٨٠ ÷ π
  1. أ) ١٫٥ × ١٨٠ ÷ ٣٫١٤١٦ = ٢٧٠ ÷ ٣٫١٤١٦ ≈ ٨٥٫٩٤٣ ≈ ٨٥٫٩°
  2. ب) ٢٫٢ × ١٨٠ ÷ ٣٫١٤١٦ = ٣٩٦ ÷ ٣٫١٤١٦ ≈ ١٢٦٫٠٤ ≈ ١٢٦°
  3. ج) ١٫٠٦ × ١٨٠ ÷ ٣٫١٤١٦ = ١٩٠٫٨ ÷ ٣٫١٤١٦ ≈ ٦٠٫٧٤٨ ≈ ٦٠٫٧°
  4. د) ١٫٩٣ × ١٨٠ ÷ ٣٫١٤١٦ = ٣٤٧٫٤ ÷ ٣٫١٤١٦ ≈ ١١٠٫٦١ ≈ ١١١°
  5. هـ) ٠٫٦٨ × ١٨٠ ÷ ٣٫١٤١٦ = ١٢٢٫٤ ÷ ٣٫١٤١٦ ≈ ٣٨٫٩٦٤ ≈ ٣٩٫٠°

٧ آلة حاسبة • ٣ أرقام معنوية

تأكد أن الآلة على وضع الراديان قبل حساب جا ١٫٢ وجتا ٠٫٨. الزوايا التي فيها π يمكن تحويلها إلى درجات إن أحببت.
تذكير: π١٠ = ١٨° ، π٥ = ٣٦° ، π٣٤ = ١٣٥° ، وظا ١٣٥° = −١ بالضبط.

π١٠ = ١٨° ، π٥ = ٣٦° ، π٣٤ = ١٣٥°. ظا ١٣٥° = −١ بالضبط.
الخطوات: اضبط الآلة على وضع الراديان للأعداد العشرية. أما الزوايا بدلالة π فيمكن تحويلها إلى درجات ثم حسابها بوضع الدرجة.
  1. أ) الآلة على راديان: جا(١٫٢) ≈ ٠٫٩٣٢٠ ≈ ٠٫٩٣٢
  2. ب) جتا(٠٫٨) ≈ ٠٫٦٩٦٧ ≈ ٠٫٦٩٧
  3. ج) ظا(١٫٢) = جا١٫٢ ÷ جتا١٫٢ ≈ ٠٫٩٣٢ ÷ ٠٫٣٦٢٤ ≈ ٢٫٥٧
  4. د) π١٠ = ١٨° ← جا ١٨° ≈ ٠٫٣٠٩
  5. هـ) π٥ = ٣٦° ← جتا ٣٦° ≈ ٠٫٨٠٩
  6. و) π٣٤ = ١٣٥° ← ظا ١٣٥° = ظا(١٨٠°−٤٥°) = −ظا ٤٥° = −١ بالضبط

٨ مثلث • قانون الجيب التمام

أطوال الضلعين المجاورين ٧ و ١٠، والزاوية المحصورة بينهما π٥١٣. أوجد طول الضلع أ مقرّبًا إلى ٣ أرقام معنوية.
قانون الجيب التمام: أ² = ب² + جـ² − ٢ب جـ جتا(الزاوية المحصورة)
أي: أ² = ٧² + ١٠² − ٢×٧×١٠ × جتا(π٥١٣)
الآلة على وضع الراديان لأن الزاوية معطاة بالراديان.

الزاوية محصورة بين الضلعين ٧ و ١٠، فالضلع المطلوب هو المقابل لها. استخدم أ² = ٧² + ١٠² − ٢×٧×١٠×جتا(π٥١٣). الآلة على وضع الراديان.
قانون الجيب التمام: أ² = ب² + جـ² − ٢ب جـ جتا أ
  1. الزاوية المحصورة بين ٧ و ١٠ هي π٥١٣، فالضلع المطلوب أ هو المقابل لها.
  2. حوّل الزاوية إن أحببت للتأكد: π٥١٣ = ٥×٣٫١٤١٦ ÷ ١٣ ≈ ١٥٫٧٠٨ ÷ ١٣ ≈ ١٫٢٠٨ راد.
  3. الآلة على وضع الراديان: جتا(١٫٢٠٨) ≈ ٠٫٣٥٤٦.
  4. أ² = ٧² + ١٠² − ٢×٧×١٠×٠٫٣٥٤٦
    = ٤٩ + ١٠٠ − ١٤٠ × ٠٫٣٥٤٦
    = ١٤٩ − ٤٩٫٦٤٤
    = ٩٩٫٣٥٦
  5. أ = √٩٩٫٣٥٦ ≈ ٩٫٩٦٧ ≈ ٩٫٩٧ (٣ أرقام معنوية)

٩ السلّم والجدار ★

سلّم طوله ٦ م يستند إلى حائط رأسي. المسافة بين قاعدة السلّم والجدار ٢٫٢٥ م. يكون استخدام السلّم آمنًا عندما لا تقل الزاوية س عن ١٫٢ راديان. أوجد س بالراديان إلى ٣ أرقام معنوية، وهل السلّم آمن؟
القوانين:
١) في المثلث القائم: جتا س = المجاور ÷ الوتر
٢) س = قوس جتا (جتا س) ، والآلة على وضع الراديان
٣) شرط الأمان: س ≥ ١٫٢ راد ← آمن ، وإلا غير آمن.

المثلث قائم عند الجدار. الوتر هو طول السلّم ٦ م، والضلع المجاور للزاوية س (عند الأرض) هو ٢٫٢٥ م.
جتا س = المجاور / الوتر
ثم س = قوس جتا(الناتج) بالراديان. قارن الناتج بشرط الأمان س ≥ ١٫٢.
الخطوة ١ — تحديد أضلاع المثلث القائم
الوتر (طول السلّم) = ٦ م
المجاور للزاوية س (المسافة عن الجدار) = ٢٫٢٥ م
المقابل = ارتفاع نقطة الاستناد على الجدار (لا نحتاجه إن استخدمنا الجتا مباشرة).
الخطوة ٢ — قانون جيب التمام في المثلث القائم (تعريف جتا)
جتا س = المجاورالوتر = ٢٫٢٥٦ = ٠٫٣٧٥
الخطوة ٣ — إيجاد س بالراديان
الآلة على وضع الراديان:
س = قوس جتا(٠٫٣٧٥) = cos⁻¹(٠٫٣٧٥) ≈ ١٫١٨٦٣٩٩٥ راد
التقريب إلى ٣ أرقام معنوية: ١٫١٩ راديان
تحقق بطريقة ثانية (اختياري)
المقابل = √(٦² − ٢٫٢٥²) = √(٣٦ − ٥٫٠٦٢٥) = √٣٠٫٩٣٧٥ ≈ ٥٫٥٦٢ م
ظا س = ٥٫٥٦٢ ÷ ٢٫٢٥ ≈ ٢٫٤٧٢ ← قوس ظا(٢٫٤٧٢) ≈ ١٫١٨٦ راد (نفس الناتج)
الخطوة ٤ — فحص شرط الأمان
الشرط: السلّم آمن إذا س ≥ ١٫٢ راد.
١٫١٩ < ١٫٢٠ لذلك الزاوية أصغر من الحد الأدنى.
الاستنتاج: السلّم غير آمن للاستخدام.
