١-٢ طول القوس

طبّق ن = نق × هـ بالراديان، ثم قرّب إلى ٣ أرقام معنوية عند الحاجة. π تُكتب يسار الرقم مثل π٣.

نتيجتك: ٠ / ٠

π القوانين

طول القوسن = نق × هـ  (هـ بالراديان)
إيجاد الزاويةهـ = ن ÷ نق
إيجاد نصف القطرنق = ن ÷ هـ
محيط القطاعمحيط = ٢نق + ن = نق(٢ + هـ)
محيط القطعة الصغرىمحيط = ن + الوتر
التحويل إلى درجاتهـ° = هـ × ١٨٠ ÷ π
ن = طول القوس نق = نصف القطر هـ = الزاوية بالراديان هـ° = الزاوية بالدرجات

اكتب الكسر في خانتين. π تكون يسار الرقم مثل π٣. التقريب إلى ٣ أرقام معنوية عند الحاجة.

تمارين ١-٢ • طول القوس

١ الزاوية الحادة من طول القوس

نصف القطر ٨ سم، وطول القوس الأصغر أ ب = ٧٫٥ سم. أوجد قياس الزاوية الحادة أ م ب.
القانون: هـ = ن ÷ نق

أ) قياس الزاوية بالراديان
ب) قياس الزاوية بالدرجات
هـ بالراديان = ن ÷ نق. ثم هـ° = هـ × ١٨٠ ÷ π.
القانون: هـ = ن ÷ نق
  1. هـ = ٧٫٥ ÷ ٨ = ٠٫٩٣٧٥ راد ≈ ٠٫٩٣٨ راديان (٣ أرقام معنوية)
  2. هـ° = ٠٫٩٣٧٥ × ١٨٠ ÷ ٣٫١٤١٦ ≈ ١٦٨٫٧٥ ÷ ٣٫١٤١٦ ≈ ٥٣٫٧°

٢ إيجاد نصف القطر من القوس

النقطتان ل ، ك على المحيط. طول القوس ل ك = ١٢ سم ، و ق(ل م ك) = ١٫٦ راد. أوجد نصف القطر.
القانون: نق = ن ÷ هـ

نصف القطر =
من ن = نق × هـ نحصل على نق = ن ÷ هـ.
  1. نق = ١٢ ÷ ١٫٦ = ٧٫٥ سم

٣ طول القوس بدلالة π

القانون: ن = نق × هـ . اكتب الناتج بدلالة π.

اضرب نصف القطر في الزاوية ثم اختصر. مثال: ١٠ × π٣/٥ = π٦.
  1. أ) ٤ × π٢ = π٢
  2. ب) ٣ × π٣٤ = π٩٤
  3. ج) ١٠ × π٣٥ = π٦
  4. د) ١٢ × π٥٦ = π١٠

٤ طول القوس بالراديان

ن = نق × هـ ثم قرّب إلى ٣ أرقام معنوية إن لزم.

أ) نق = ١٢ سم ، هـ = ١٫٢ راد
ب) نق = ٤٫٥ سم ، هـ = ٠٫٥٥ راد
اضرب مباشرة: ١٢ × ١٫٢ و ٤٫٥ × ٠٫٥٥.
  1. أ) ١٢ × ١٫٢ = ١٤٫٤ سم
  2. ب) ٤٫٥ × ٠٫٥٥ = ٢٫٤٧٥ ≈ ٢٫٤٨ سم

٥ إيجاد الزاوية من القوس

هـ = ن ÷ نق

أ) نق = ٢٠ سم ، القوس = ٥ سم
ب) نق = ٩٫٣٦ سم ، القوس = ٥٫٢ سم
اقسم طول القوس على نصف القطر.
  1. أ) ٥ ÷ ٢٠ = ٠٫٢٥ راد
  2. ب) ٥٫٢ ÷ ٩٫٣٦ ≈ ٠٫٥٥٥٥٦ ≈ ٠٫٥٥٦ راد

٦ العلاقة ن = نق × هـ

القوس الأصغر ل ك طوله ن يقابل الزاوية هـ. من ن = نق × هـ أوجد المطلوب.

ن = نق × هـ . انقل المجهول بقسمة الطرفين.
  1. أ) ن = ٩ × π٣٥ = π٢٧٥
  2. ب) نق = ١٤٫٩ ÷ ٢٫٩٨ = ٥ سم
  3. ج) هـ = (π٤٩٩) ÷ ٧ = π٤٩٦٣ = π٧٩

٧ محيط القطاع → الزاوية

نق = ١٥ سم ، محيط القطاع = ٥٠ سم. أوجد هـ.
القانون: محيط القطاع = ٢نق + نق·هـ

هـ =
٥٠ = ٢×١٥ + ١٥هـ ثم حلّ المعادلة.
  1. ٥٠ = ٣٠ + ١٥هـ
  2. ١٥هـ = ٢٠
  3. هـ = ٢٠÷١٥ = ٤/٣ أو ١٫٣٣ راد

٨ محيط القطاع → نصف القطر

ق(أ م ب) = ٢٫٣ راد ، محيط القطاع = ٣٠ سم. أوجد م أ.
القانون: ٣٠ = نق(٢ + ٢٫٣)

م أ =
نق = ٣٠ ÷ (٢ + ٢٫٣) = ٣٠ ÷ ٤٫٣
  1. ٣٠ = نق × ٤٫٣
  2. نق = ٣٠ ÷ ٤٫٣ ≈ ٦٫٩٧٦٧ ≈ ٦٫٩٨ سم

٩ محيط القطعة الصغرى

نق = ٢٥ سم ، الوتر أ ب = ١٤ سم. محيط القطعة الصغرى = طول القوس الأصغر + الوتر.
أولًا أوجد الزاوية من جا(هـ/٢) = (الوتر/٢) ÷ نق

محيط القطعة الصغرى ≈
جا(هـ/٢) = ٧÷٢٥ = ٠٫٢٨ ثم هـ = ٢ قوس جا(٠٫٢٨) ثم ن = ٢٥هـ ثم أضف ١٤.
  1. جا(هـ/٢) = ٧ ÷ ٢٥ = ٠٫٢٨
  2. هـ/٢ ≈ ٠٫٢٨٣٨ راد ، إذن هـ ≈ ٠٫٥٦٧٦ راد
  3. ن = ٢٥ × ٠٫٥٦٧٦ ≈ ١٤٫١٩ سم
  4. المحيط = ن + الوتر = ١٤٫١٩ + ١٤ ≈ ٢٨٫٢ سم
